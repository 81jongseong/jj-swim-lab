```typescript
// 1️⃣ 회원가입 시 선택
interface SignupFlow {
  step1: '가입 유형 선택' {
    options: [
      '개인 회원 (자유수영)',      // → 직접 결제
      '센터 회원 (소속 센터 있음)' // → 센터 선택
    ]
  }
  
  step2_개인: '결제 정보 입력' {
    plan: 'premium' // ₩19,900/월
    paymentMethod: '카드/계좌'
  }
  
  step2_센터: '센터 선택' {
    centerCode: '입력'
    centerVerification: '승인 대기'
  }
}

// 2️⃣ 과금 대상 결정
function getBillingTarget(user: User): 'user' | 'center' {
  if (user.centerId && user.centerApproved) {
    // 센터 소속 + 승인됨 → 센터 과금
    return 'center';
  }
  
  // 그 외 → 회원 직접 과금
  return 'user';
}

// 3️⃣ 월 사용료 계산
function calculateMonthlyFee(user: User): {
  target: 'user' | 'center';
  amount: number;
  currency: 'KRW';
} {
  const billingTarget = getBillingTarget(user);
  
  if (billingTarget === 'center') {
    // 센터 과금: 센터 패키지에 포함
    return {
      target: 'center',
      amount: 0, // 회원은 무료
      currency: 'KRW'
    };
  }
  
  // 회원 직접 과금
  return {
    target: 'user',
    amount: 19900,
    currency: 'KRW'
  };
}
```
