```typescript
interface SubstituteRequest {
  requestId: string;

  centerId: string;
  
  // 수업 정보
  courseId: string;
  scheduledDate: Date;
  duration: number;             // 수업 시간 (분)
  studentCount: number;         // 수강생 수
  
  // 요청 사유
  reason: 'sick_leave' | 'vacation' | 'emergency' | 'training' | 'other';
  detailedReason: string;
  urgency: 'low' | 'medium' | 'high' | 'emergency';
  
  // 대리 강사 조건
  requiredLevel: 'trainee' | 'junior' | 'senior' | 'master';
  preferredInstructors?: string[]; // 선호 대리 강사
  
  // 상태
  status: 'pending' | 'assigned' | 'confirmed' | 'completed' | 'cancelled';
  createdAt: Date;
}
```

## 🎪 **게스트 강습 유형별 관리**

### **🚑 긴급 대리 (Emergency Substitute)**
- **상황**: 당일 갑작스런 결석
- **수당**: 기본 수당의 150%
- **조건**: 2시간 내 배정 완료
- **대상**: 같은 센터 또는 인근 센터 강사

### **🏖️ 계획 대리 (Planned Substitute)**
- **상황**: 미리 계획된 휴가/출장
- **수당**: 기본 수당의 120%
- **조건**: 1주일 전 요청
- **대상**: 네트워크 내 모든 강사

### **🎓 특별 강습 (Special Program)**
- **상황**: 특수 프로그램 (마스터클래스 등)
- **수당**: 프로그램별 차등
- **조건**: 특정 자격 요구
- **대상**: 해당 분야 전문 강사

### **🔄 정기 순환 (Regular Rotation)**
- **상황**: 센터 간 정기 교류
- **수당**: 협의된 고정 수당
- **조건**: 월 단위 계약
- **대상**: 시니어 이상 강사

```typescript
function findBestSubstituteInstructor(request: SubstituteRequest): string[] {
  const candidates = instructors.filter(instructor => 
    instructor.level >= request.requiredLevel &&
    instructor.isAvailable(request.scheduledDate) &&
    instructor.specialties.includes(request.courseType) &&
    getDistance(instructor.location, request.centerLocation) <= 30 // 30km 이내
  );
  
  return candidates
    .sort((a, b) => calculateMatchScore(b, request) - calculateMatchScore(a, request))
    .slice(0, 5); // 상위 5명 추천
}
```
