## 🔄 **자동 등급 계산 시스템**

### **📊 종합 점수 계산**
```typescript
function calculateCenterGrade(kpis: {
  operational: OperationalKPI;
  quality: QualityKPI;
  safety: SafetyKPI;
  education: EducationKPI;
}): string {
  
  // 각 영역별 점수 계산 (0-100점)
  const operationalScore = calculateOperationalScore(kpis.operational);
  const qualityScore = calculateQualityScore(kpis.quality);
  const safetyScore = calculateSafetyScore(kpis.safety);
  const educationScore = calculateEducationScore(kpis.education);
  
  // 가중 평균 계산
  const totalScore = 
    (operationalScore * 0.4) +
    (qualityScore * 0.3) +
    (safetyScore * 0.2) +
    (educationScore * 0.1);
    
  // 등급 결정
  if (totalScore >= 90) return 'Platinum';
  if (totalScore >= 75) return 'Gold';
  if (totalScore >= 60) return 'Silver';
  return 'Bronze';
}
```
